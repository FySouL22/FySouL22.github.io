#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
PyPort-Scanner Pro v2.5 by Mohamed Fathi (Ft7y.Sec)
GitHub: https://github.com/FySouL22/pyport-scanner-pro
"""
import socket
import sys
import json
import argparse
from concurrent.futures import ThreadPoolExecutor, as_completed
from datetime import datetime

COMMON_PORTS = {
    21: "FTP", 22: "SSH", 23: "Telnet", 25: "SMTP", 53: "DNS",
    80: "HTTP", 110: "POP3", 135: "MSRPC", 139: "NetBIOS",
    143: "IMAP", 443: "HTTPS", 445: "SMB", 3306: "MySQL",
    3389: "RDP", 5432: "PostgreSQL", 8080: "HTTP-Proxy"
}

def grab_banner(target_ip, port, timeout=1.0):
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            s.settimeout(timeout)
            s.connect((target_ip, port))
            if port in [80, 8080, 443]:
                s.sendall(b"HEAD / HTTP/1.1\r\nHost: " + target_ip.encode() + b"\r\n\r\n")
            else:
                s.sendall(b"\r\n")
            raw = s.recv(1024).decode('utf-8', errors='ignore').strip()
            return raw.splitlines()[0][:80] if raw else "Service Active"
    except Exception:
        return COMMON_PORTS.get(port, "Unknown")

def scan_port(target_ip, port, timeout=0.8):
    try:
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
            s.settimeout(timeout)
            if s.connect_ex((target_ip, port)) == 0:
                return {
                    "port": port,
                    "service": COMMON_PORTS.get(port, "Unknown"),
                    "banner": grab_banner(target_ip, port, timeout)
                }
    except Exception:
        pass
    return None

def main():
    parser = argparse.ArgumentParser(description="PyPort-Scanner Pro by Ft7y.Sec")
    parser.add_argument("-t", "--target", required=True, help="Target Host / IP")
    parser.add_argument("-p", "--ports", default="1-1024", help="Port range (e.g. 1-1024, 80,443, top20)")
    parser.add_argument("-w", "--workers", type=int, default=50, help="Thread count (default: 50)")
    parser.add_argument("-o", "--output", help="Save to JSON file")
    args = parser.parse_args()

    target_ip = socket.gethostbyname(args.target)
    print(f"[*] Scanning {args.target} ({target_ip}) with {args.workers} workers...")
    
    ports = range(1, 1025)
    open_ports = []
    with ThreadPoolExecutor(max_workers=args.workers) as pool:
        futures = {pool.submit(scan_port, target_ip, p): p for p in ports}
        for f in as_completed(futures):
            res = f.result()
            if res:
                open_ports.append(res)
                print(f"[+] Port {res['port']:5d}/TCP OPEN ({res['service']}) -> {res['banner']}")

if __name__ == "__main__":
    main()
