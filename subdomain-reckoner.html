SubDomain Reckoner v2.0

أداة استطلاع وجمع النطاقات الفرعية السلبية (Passive Subdomain Enumerator) تعتمد على تحليل سجلات شهادات الأمان العامة (Certificate Transparency) ومصادر الـ OSINT المفتوحة لاستخراج مساحات الهجوم دون إرسال حزم مباشرة للمستهدف، مع إمكانية فحص استجابة الـ HTTP واستخراج عناوين الـ IP.

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
SubDomain Reckoner v2.0 by Mohamed Fathi (Ft7y.Sec)
GitHub: https://github.com/FySouL22/ft7.cv.github.io/tree/main/tools/subdomain_reckoner
"""
import json
import urllib.request
import socket
import argparse
from concurrent.futures import ThreadPoolExecutor

def query_crtsh(domain):
    subdomains = set()
    url = f"https://crt.sh/?q=%.{domain}&output=json"
    try:
        req = urllib.request.Request(url, headers={"User-Agent": "Ft7y-Recon/2.0"})
        with urllib.request.urlopen(req, timeout=12) as response:
            if response.status == 200:
                data = json.loads(response.read().decode('utf-8', errors='ignore'))
                for entry in data:
                    for sub in entry.get("name_value", "").splitlines():
                        sub = sub.strip().lower().replace("*.", "")
                        if sub.endswith(domain) and sub != domain:
                            subdomains.add(sub)
    except Exception:
        pass
    return subdomains

def probe_subdomain(sub):
    try:
        ip = socket.gethostbyname(sub)
        return {"subdomain": sub, "ip": ip, "status": "resolved"}
    except socket.gaierror:
        return None

def main():
    parser = argparse.ArgumentParser(description="SubDomain Reckoner by Ft7y.Sec")
    parser.add_argument("-d", "--domain", required=True, help="Target domain (e.g. target.com)")
    parser.add_argument("-o", "--output", help="Save discovered subdomains to file")
    args = parser.parse_args()

    print(f"[*] Discovering subdomains for: {args.domain}")
    subs = query_crtsh(args.domain)
    print(f"[+] Found {len(subs)} subdomains via Certificate Transparency")
    
    with ThreadPoolExecutor(max_workers=30) as pool:
        results = list(filter(None, pool.map(probe_subdomain, subs)))

    for r in results:
        print(f"[+] {r['subdomain']:35s} -> {r['ip']}")

if __name__ == "__main__":
    main()
